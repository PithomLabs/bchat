# SingleHop (~17 min, 64 questions)

BENCHMARK_LONGMEMEVAL=true OPENROUTER_API_KEY=$OPENROUTER_API_KEY \

# Abstention (~3 min, 12 questions)
BENCHMARK_LONGMEMEVAL=true OPENROUTER_API_KEY=$OPENROUTER_API_KEY \

# Aggregate summary (run after all types)
go test ./server/router/api/v1/agent/ -run TestBenchmarkAggregate -v -count=1
